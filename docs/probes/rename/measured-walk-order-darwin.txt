# Darwin 25.6.0 arm64, uid 501

## when is each pathname copied in
# getname()/copyinstr on the destination, against sources that fail at different phases
control: both absent, short        rename(b'nope', b'alsonope') -> rc=-1 errno=ENOENT
control: over-long source          rename(b'zzzzzzzzzzzz..(5000)', b'alsonope') -> rc=-1 errno=ENAMETOOLONG
free source name, over-long dest   rename(b'nope', b'zzzzzzzzzzzz..(5000)') -> rc=-1 errno=ENOENT
free source name, mid-length dest  rename(b'nope', b'zzzzzzzzzzzz..(2000)') -> rc=-1 errno=ENOENT
existing source, over-long dest    rename(b'f', b'zzzzzzzzzzzz..(5000)') -> rc=-1 errno=ENAMETOOLONG
DISCRIMINATOR: source parent absent rename(b'nodir/kid', b'zzzzzzzzzzzz..(5000)') -> rc=-1 errno=ENOENT
DISCRIMINATOR: source parent unsearchable rename(b'nosearch/kid', b'zzzzzzzzzzzz..(5000)') -> rc=-1 errno=EACCES
DISCRIMINATOR: source parent is a file rename(b'f/kid', b'zzzzzzzzzzzz..(5000)') -> rc=-1 errno=ENOTDIR

## and the same question for EFAULT
# an unreadable pathname pointer, which getname() reports the same way it reports length
control: bogus source              rename(<unmapped>, b'alsonope') -> rc=-1 errno=EFAULT
absent source, bogus dest          rename(b'nope', <unmapped>) -> rc=-1 errno=ENOENT
over-long source, bogus dest       rename(b'zzzzzzzzzzzz..(5000)', <unmapped>) -> rc=-1 errno=ENAMETOOLONG
existing source, bogus dest        rename(b'f', <unmapped>) -> rc=-1 errno=EFAULT

## how much of the source is settled first
# destination is always b'f/x', which answers ENOTDIR on its own
free name                          rename(b'nope', b'f/x') -> rc=-1 errno=ENOENT
regular file (control)             rename(b'f', b'f/x') -> rc=-1 errno=ENOTDIR
a directory                        rename(b'dir', b'f/x') -> rc=-1 errno=ENOTDIR
a directory, trailing sep          rename(b'dir/', b'f/x') -> rc=-1 errno=ENOTDIR
symlink to a directory             rename(b'ld', b'f/x') -> rc=-1 errno=ENOTDIR
symlink to a file                  rename(b'lf', b'f/x') -> rc=-1 errno=ENOTDIR
dangling symlink                   rename(b'dangling', b'f/x') -> rc=-1 errno=ENOTDIR
300-byte final component           rename(b'zzzzzzzzzzzz..(300)', b'f/x') -> rc=-1 errno=ENAMETOOLONG
under an unsearchable parent       rename(b'nosearch/kid', b'f/x') -> rc=-1 errno=EACCES
parent absent                      rename(b'nodir/kid', b'f/x') -> rc=-1 errno=ENOENT
parent is a regular file           rename(b'f/kid', b'f/x') -> rc=-1 errno=ENOTDIR

## which navigation the source settles first
# destination is always b'f/x'; only the bare root is early
the root                           rename(b'/', b'f/x') -> rc=-1 errno=EISDIR
the root as /.                     rename(b'/.', b'f/x') -> rc=-1 errno=ENOTDIR
the root as /..                    rename(b'/..', b'f/x') -> rc=-1 errno=ENOTDIR
the root as /dev/..                rename(b'/dev/..', b'f/x') -> rc=-1 errno=ENOTDIR
. here                             rename(b'.', b'f/x') -> rc=-1 errno=ENOTDIR
.. here                            rename(b'..', b'f/x') -> rc=-1 errno=ENOTDIR
dir/.                              rename(b'dir/.', b'f/x') -> rc=-1 errno=ENOTDIR
dir/..                             rename(b'dir/..', b'f/x') -> rc=-1 errno=ENOTDIR
dir/sub/..                         rename(b'dir/sub/..', b'f/x') -> rc=-1 errno=ENOTDIR

## an orphaned destination parent, against the final lookups
# the cwd is an rmdir'd orphan, so a relative destination's parent is one
control: short destination name    rename(b'/private/var..(70)', b'x') -> rc=-1 errno=ENOENT
DISCRIMINATOR: 300-byte destination name rename(b'/private/var..(70)', b'zzzzzzzzzzzz..(300)') -> rc=-1 errno=ENAMETOOLONG
300-byte *source* final name       rename(b'/private/var..(369)', b'x') -> rc=-1 errno=ENAMETOOLONG
absent absolute source             rename(b'/private/var..(73)', b'x') -> rc=-1 errno=ENOENT
